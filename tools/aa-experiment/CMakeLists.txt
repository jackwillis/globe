cmake_minimum_required(VERSION 3.20)
project(pe_aa_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AA_SANITIZE "Instrument experiment correctness checks" OFF)
get_filename_component(GLOBE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT GLOBE_CORE_BUILD)
  message(FATAL_ERROR "Set GLOBE_CORE_BUILD to a matching baseline core build directory")
endif()
file(STRINGS "${GLOBE_CORE_BUILD}/CMakeCache.txt" CORE_SOURCE REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REPLACE "CMAKE_HOME_DIRECTORY:INTERNAL=" "" CORE_SOURCE "${CORE_SOURCE}")
if(NOT CORE_SOURCE STREQUAL GLOBE_ROOT)
  message(FATAL_ERROR "The core build belongs to another worktree; create a private matching build")
endif()
add_library(aa_candidate aa.cpp)
target_include_directories(aa_candidate PUBLIC "${GLOBE_ROOT}/include")
target_compile_options(aa_candidate PUBLIC -fno-exceptions -fno-rtti PRIVATE -Wall -Wextra -Wpedantic)
add_library(baseline_core STATIC IMPORTED)
set_target_properties(baseline_core PROPERTIES IMPORTED_LOCATION "${GLOBE_CORE_BUILD}/libglobe_core.a")
if(AA_SANITIZE)
  target_compile_options(aa_candidate PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(aa_candidate PUBLIC -fsanitize=address,undefined)
endif()
add_executable(aa_tests test_aa.cpp)
target_link_libraries(aa_tests PRIVATE aa_candidate baseline_core)
add_executable(aa_fixtures fixtures.cpp)
target_link_libraries(aa_fixtures PRIVATE aa_candidate baseline_core)
enable_testing()
add_test(NAME aa_synthetic COMMAND aa_tests)
add_executable(aa_synthetic_driver synthetic_driver.cpp)
target_link_libraries(aa_synthetic_driver PRIVATE aa_candidate baseline_core)
add_executable(aa_extractor_audit extractor_audit.cpp)
target_link_libraries(aa_extractor_audit PRIVATE aa_candidate baseline_core)
add_executable(aa_feature_sidecar feature_sidecar.cpp)
target_link_libraries(aa_feature_sidecar PRIVATE aa_candidate baseline_core)
